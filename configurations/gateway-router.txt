<b>Gateway Router config</b>:
enable
configure terminal
hostname Gateway
interface fa0/0
no shutdown
interface fa0/0.2
encapsulation dot1q 2
ip address 192.168.1.1 255.255.255.0
no shutdown
interface fa0/0.3
encapsulation dot1q 3
ip address 192.168.0.1 255.255.255.0
no shutdown
interface se0/3/0
ip address 16.245.239.1 255.255.255.252
no shutdown

!Telephony service configuration
telephony-service
max-ephones 5
max-dn 5
ip source-address 192.168.0.1 port 2000
auto assign 1 to 5
exit

!IP Phone dial number
ephone-dn  1 
  number 1001
  exit
  ephone-dn 2
  number 2
  number 1002
  exit

!DCHP config for voice LAN
ip dhcp excluded-address 192.168.0.1 192.168.0.10
ip dhcp pool voiceLAN
  network 192.168.0.0 255.255.255.0
  default-router 192.168.0.1
  option 150 ip 192.168.0.1
  domain cisco
  dns 1.1.1.1
  exit

!DHCP config for LAN
ip dhcp excluded-address 192.168.1.1 192.168.1.10
ip dhcp pool lan
  network 192.168.1.0 255.255.255.0
  default-router 192.168.1.1
  domain cisco
  dns 1.1.1.1
  exit

!default route configuration
ip route 0.0.0.0 0.0.0.0 16.245.239.2
